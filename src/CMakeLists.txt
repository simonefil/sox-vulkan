# Every shader under vulkan/ goes through the same two steps: glslc compiles
# it to a .spv, then sox_embed_binary turns that into a C array in a _spv.inc
# the host includes.  Nothing is loaded from disk at run time, which is what
# lets a single executable carry all the profiles.
#
# The three names per shader -- _SOURCE, _SPV, _INC -- are spelled out rather
# than generated by a loop because each pair is a separate add_custom_command
# with its own DEPENDS, and the reference-profile shaders additionally depend
# on double_double.glsl, which glslc resolves relative to the including file.
#
# sox_embed_binary is built for the host and must exist before any .inc, hence
# the DEPENDS on the target; it is trivial enough that cross-compiling has not
# needed a prebuilt copy so far.

# at configure time, so a build from a tarball or with no git simply has no

# case here.
find_package(Git QUIET)

    execute_process(
        COMMAND "${GIT_EXECUTABLE}" rev-parse --short=8 HEAD
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}"
        OUTPUT_VARIABLE SOX_GIT_COMMIT
        OUTPUT_STRIP_TRAILING_WHITESPACE
        ERROR_QUIET
    )
    if(SOX_GIT_COMMIT)
        target_compile_definitions(sox_exe PRIVATE SOX_GIT_COMMIT="${SOX_GIT_COMMIT}")
    endif()
endif()

    OUTPUT_NAME sox

# glslang_dynamic.c resolves the DLL by name at run time rather than linking
# it, so nothing puts it next to the executable; these two steps do, for the
# build tree and for an install.  Without them a Vulkan effect fails at the
# point VkFFT first needs to compile a shader, not at load.
if(HAVE_VULKAN AND IS_WINDOWS)
    add_custom_command(TARGET sox_exe POST_BUILD
        COMMAND "${CMAKE_COMMAND}" -E copy_if_different
            "${GLSLANG_RUNTIME_DLL}" "$<TARGET_FILE_DIR:sox_exe>"
        VERBATIM
    )
